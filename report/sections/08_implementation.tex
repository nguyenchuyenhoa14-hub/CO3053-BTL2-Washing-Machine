\section{Software Implementation \& MISRA-C:2012 Compliance}
\label{sec:impl}
% =============================================================================

Only the key architectural fragments are presented in this section; the unabridged, line-by-line
source code listings are compiled in Appendix~\ref{app:source}. The entire codebase is implemented in
strict ISO C99, compiled with aggressive compiler diagnostics enabled:
\begin{center}
  \code{-Wall -Wextra -Werror -pedantic -std=c99}
\end{center}
on both the host x86\_64 toolchain (`gcc`) and the embedded cross-compiler
(\code{arm-none-eabi-gcc -mcpu=cortex-m7 -Os}). The production firmware utilizes zero dynamic heap
allocation (\code{malloc}/\code{free}), avoids bloated standard library formatters (no \code{printf}),
and occupies only $9.3\,\text{KB}$ of internal Flash memory.

\subsection{Formal MISRA-C:2012 Compliance Matrix}
\label{sec:misra}

In mission-critical embedded control, adherence to the Motor Industry Software Reliability Association
(MISRA-C:2012) guidelines is paramount to eliminate undefined runtime behaviors, memory corruption,
and unhandled exceptions \cite{misra2012}. Table~\ref{tab:misra} documents the primary rules enforced
throughout our implementation.

\begin{table}[H]
  \centering
  \caption{MISRA-C:2012 compliance verification and implementation evidence}
  \label{tab:misra}
  \small
  \begin{tabularx}{\linewidth}{lL{4.1cm}Y}
    \toprule
    \textbf{MISRA Rule} & \textbf{Formal Guideline Requirement} & \textbf{Implementation Enforcement in Washing Machine Codebase} \\
    \midrule
    \textbf{Rule 2.1} & A project shall not contain unreachable code. & Audited and verified: all redundant defensive guards that could never be reached via valid transitions were eliminated from the core FSM. \\
    \textbf{Rule 8.7} & Functions and objects shall be declared with internal linkage (\code{static}) where possible. & All internal transition handlers (\code{handle_coin_deposit}, \code{enter_running}, \code{handle_stop_button}) and file-scope buffers are strictly declared \code{static}. \\
    \textbf{Rule 8.9} & An object should be defined at block scope if its identifier only appears within a single function. & File-scope globals are banned in the core FSM; all operational state is encapsulated within an explicit, passed context pointer \code{wm_context_t *ctx}. \\
    \textbf{Rule 11.4} & A conversion should not be performed between a pointer to object type and a different pointer. & Zero type punning or void-pointer casting; all callback structures and event objects are strongly and explicitly typed. \\
    \textbf{Rule 12.4} & Evaluation of integer expressions shall not lead to wrap-around or undefined overflow. & Explicit integer arithmetic guards: coin deposit checks against $\code{UINT32_MAX} - v$; phase timer arithmetic guarded by modulo boundaries. \\
    \textbf{Rule 17.2} & Functions shall not call themselves, either directly or indirectly (Zero Recursion). & The FSM event dispatcher uses flat iterative \code{switch-case} constructs. Stack depth is deterministic and statically bounded to $<128\,\text{bytes}$. \\
    \textbf{Rule 21.3} & Dynamic memory allocation functions of \code{<stdlib.h>} shall not be used. & Absolute ban on \code{malloc}, \code{calloc}, \code{realloc}, and \code{free}. All data structures, buffers, and FSM contexts are statically or automatically allocated. \\
    \bottomrule
  \end{tabularx}
\end{table}

\subsection{State, Event, and Context Encapsulation}

\begin{lstlisting}[caption={State, event, and context definitions (\code{washing_machine_fsm.h})}]
typedef enum {
    WM_STATE_STANDBY = 0,   /* RLED=ON,  BLED=OFF                     */
    WM_STATE_COLLECTING,    /* 0 < deposit < 50c; RLED=ON, BLED=OFF   */
    WM_STATE_READY,         /* deposit >= 50c;    RLED=OFF, BLED=ON   */
    WM_STATE_RUNNING,       /* BLED blink 1 Hz, timer ticking         */
    WM_STATE_PAUSED,        /* actuators stopped, timer STILL ticking */
    WM_STATE_ERROR          /* RLED blink 2 Hz, actuators stopped     */
} wm_state_t;

typedef enum {
    WM_EVT_NONE = 0, WM_EVT_COIN_10, WM_EVT_COIN_20, WM_EVT_COIN_50,
    WM_EVT_BTN_RUN, WM_EVT_BTN_PAUSE, WM_EVT_BTN_STOP,
    WM_EVT_TIMER_TICK_1S, WM_EVT_TIMER_TICK_1MS,
    WM_EVT_FAULT_OCCURRED, WM_EVT_FAULT_CLEARED
} wm_event_t;

typedef struct {
    wm_state_t state;
    uint32_t coin_balance_cents;      /* b */
    uint32_t remaining_cycle_sec;     /* t */
    uint32_t stop_press_count;        /* n */
    uint32_t stop_window_timer_ms;    /* w */
    uint32_t active_error_flags;      /* f */
    hal_output_callbacks_t callbacks;
    uint32_t cycle_duration_setting;  /* T = 1800 s */
    wm_state_t state_before_error;
} wm_context_t;
\end{lstlisting}

\noindent All operational data reside in an explicitly typed \code{wm_context_t} instance.
There are zero global variables within the FSM library, enabling full reentrancy and side-by-side
concurrency across multiple automated test harnesses.

\subsection{Deterministic Event Dispatcher}

The core FSM is implemented as a single-entry function \code{wm_fsm_dispatch_event()}.
State transitions invoke dedicated \code{enter_xxx()} functions that apply the Moore output
rules in a centralized, unforgeable location.

\begin{lstlisting}[caption={Event dispatcher implementation (\code{washing_machine_fsm.c})}]
bool wm_fsm_dispatch_event(wm_context_t *ctx, wm_event_t event) {
    if (!ctx) return false;

    /* Global emergency transition: has absolute priority over every state (T17) */
    if (event == WM_EVT_FAULT_OCCURRED) {
        if (ctx->active_error_flags == WM_FAULT_NONE)
            ctx->active_error_flags = WM_FAULT_DOOR_OPEN;
        if (ctx->state != WM_STATE_ERROR) { enter_error(ctx); return true; }
        return false;
    }
    if (event == WM_EVT_TIMER_TICK_1MS) { wm_fsm_tick_1ms(ctx); return true; }

    switch (ctx->state) {
    case WM_STATE_READY:
        switch (event) {
        case WM_EVT_COIN_10:  return handle_coin_deposit(ctx, 10);
        case WM_EVT_COIN_20:  return handle_coin_deposit(ctx, 20);
        case WM_EVT_COIN_50:  return handle_coin_deposit(ctx, 50);
        case WM_EVT_BTN_RUN:  enter_running(ctx, false); return true;   /* T08 */
        case WM_EVT_BTN_STOP: return handle_stop_button(ctx);
        default:              return false;                             /* ignored */
        }
    case WM_STATE_RUNNING:
        switch (event) {
        case WM_EVT_BTN_PAUSE:     enter_paused(ctx); return true;      /* T09 */
        case WM_EVT_BTN_STOP:      return handle_stop_button(ctx);
        case WM_EVT_TIMER_TICK_1S: return handle_1s_timer_tick(ctx);    /* T10/T11 */
        default:                   return false;
        }
    case WM_STATE_PAUSED:
        switch (event) {
        case WM_EVT_BTN_RUN:       enter_running(ctx, true); return true; /* T14 resume */
        case WM_EVT_BTN_STOP:      return handle_stop_button(ctx);
        case WM_EVT_TIMER_TICK_1S: return handle_1s_timer_tick(ctx);    /* T15: timer continues */
        default:                   return false;
        }
    default:
        return false;
    }
}
\end{lstlisting}

\subsection{Coin Accumulation and Unconditional Zero Refund}

\begin{lstlisting}[caption={Safe coin accumulation and cycle initiation}]
static bool handle_coin_deposit(wm_context_t *ctx, uint32_t amount) {
    if (ctx->state == WM_STATE_STANDBY || ctx->state == WM_STATE_COLLECTING) {
        if (ctx->coin_balance_cents <= (UINT32_MAX - amount))   /* MISRA 12.4 overflow guard */
            ctx->coin_balance_cents += amount;
        if (ctx->coin_balance_cents >= WM_COIN_THRESHOLD_CENTS)
            enter_ready(ctx);                  /* T02/T04: RLED OFF, BLED ON */
        else
            enter_collecting(ctx);             /* T01/T03 */
        return true;
    } else if (ctx->state == WM_STATE_READY) {
        if (ctx->coin_balance_cents <= (UINT32_MAX - amount))
            ctx->coin_balance_cents += amount; /* T07: surplus accepted */
        return true;
    }
    return false;                              /* rejected in RUNNING/PAUSED/ERROR */
}

static void enter_running(wm_context_t *ctx, bool is_resuming) {
    ctx->state = WM_STATE_RUNNING;
    ctx->stop_press_count = 0;
    ctx->stop_window_timer_ms = 0;
    if (!is_resuming) {
        ctx->coin_balance_cents = 0;           /* clear money, zero refund (FR-03) */
        ctx->remaining_cycle_sec = ctx->cycle_duration_setting;   /* 1800 s */
    }
    CALL_HAL(&ctx->callbacks, set_rled, HAL_LED_OFF);
    CALL_HAL(&ctx->callbacks, set_bled, HAL_LED_BLINK_1HZ);
    update_running_actuators(ctx);             /* lock door, AGITATE or SPIN */
}
\end{lstlisting}

\subsection{Actuator Safety Interlock Invariant Code}
\label{sec:actuator-code}

To verify Safety Invariants $I_1$ and $I_2$ (Section~\ref{sec:invariants}) at runtime, the HAL
implements the defensive validator \code{hal_actuator_is_safe()}:

\begin{lstlisting}[caption={Runtime actuator safety invariant validator (\code{hal_led_blinker.c})}]
bool hal_actuator_is_safe(const hal_actuator_state_t *act) {
    if (act == NULL) {
        return false;
    }
    /* Invariant 1: Water inlet valve and drain pump cannot be energized simultaneously */
    if (act->water_valve_open && act->drain_pump_on) {
        return false;
    }
    /* Invariant 2: Drum motor rotation requires electromechanical door deadbolt locked */
    if ((act->motor != HAL_MOTOR_OFF) && (!act->door_locked)) {
        return false;
    }
    return true;
}
\end{lstlisting}

\subsection{Double-Press STOP and Persistent Countdown Handlers}

\begin{lstlisting}[caption={Double-press STOP evaluation and 1-second down-counter}]
static bool handle_stop_button(wm_context_t *ctx) {
    if (ctx->state == WM_STATE_STANDBY) return false;
    if (ctx->stop_press_count == 0) {          /* T05/T12: first press */
        ctx->stop_press_count = 1;
        ctx->stop_window_timer_ms = WM_DOUBLE_PRESS_WINDOW_MS;   /* 1500 ms */
        return true;
    }
    /* T06/T13: second press inside 1500 ms window -> confirmed abort */
    ctx->stop_press_count = 0;
    ctx->stop_window_timer_ms = 0;
    if ((ctx->state == WM_STATE_READY || ctx->state == WM_STATE_COLLECTING) &&
        ctx->coin_balance_cents > 0)
        CALL_HAL(&ctx->callbacks, return_coins, ctx->coin_balance_cents);
    ctx->coin_balance_cents = 0;
    enter_standby(ctx);                        /* all actuators OFF, RLED ON */
    return true;
}

static bool handle_1s_timer_tick(wm_context_t *ctx) {
    if (ctx->remaining_cycle_sec > 1) {
        ctx->remaining_cycle_sec--;                       /* T10 / T15 */
        if (ctx->state == WM_STATE_RUNNING)
            update_running_actuators(ctx);                /* AGITATE -> SPIN at 300 s */
        return true;
    } else if (ctx->remaining_cycle_sec == 1) {
        ctx->remaining_cycle_sec = 0;                     /* T11 / T16 */
        CALL_HAL_VOID(&ctx->callbacks, on_cycle_complete);
        enter_standby(ctx);
        return true;
    }
    return false;
}
\end{lstlisting}

\subsection{Debounce Integration and Non-Preemptive Super-Loop}

\begin{lstlisting}[caption={Integrating debouncer and bare-metal super-loop}]
void hal_button_update(hal_button_t *btn, bool raw_pin_high, uint32_t delta_ms) {
    bool pressed = btn->active_low ? !raw_pin_high : raw_pin_high;
    if (pressed != btn->debounced_state) {
        btn->debounce_counter += delta_ms;               /* integrate stable level */
        if (btn->debounce_counter >= HAL_DEBOUNCE_DEFAULT_MS) {   /* 30 ms threshold */
            btn->debounced_state = pressed;
            btn->debounce_counter = 0;
            if (pressed) btn->pressed_event = true;      /* latched edge */
            else         btn->released_event = true;
        }
    } else {
        btn->debounce_counter = 0;                       /* mechanical glitch: reset */
    }
}

int main(void) {
    board_init();                                     /* clocks 400 MHz, GPIO, SysTick */
    wm_fsm_init(&g_fsm, &cbs);                        /* initial state: STANDBY */
    lcd_init();
    for (;;) {
        uint32_t now = board_millis();
        while (processed != now) { processed++; process_1ms(&sec_acc_ms); }
        build_view(&view);
        if (!ui_view_equal(&view, &shown)) { ui_render(&view); lcd_flush(); shown = view; }
    }
}
\end{lstlisting}
