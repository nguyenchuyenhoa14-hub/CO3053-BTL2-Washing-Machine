\subsection{Coin processing}

\begin{figure}[H]
  \centering
  \resizebox{0.9\linewidth}{!}{%
  \begin{tikzpicture}[node distance=0.55cm]
    \node[fcterm] (s) {COIN($v$) event};
    \node[fcdec, below=of s] (idle) {state $\in$\\\{STANDBY,\\COLLECTING\}?};
    \node[fcdec, right=1.6cm of idle] (rdy) {state =\\READY?};
    \node[fcterm, right=1.4cm of rdy] (rej) {ignore coin\\\scriptsize RUNNING/PAUSED/ERROR};
    \node[fcproc, below=1.0cm of rdy] (add2) {$b \mathrel{+}= v$\\\scriptsize surplus, no refund later};
    \node[fcproc, below=1.0cm of idle] (add) {$b \mathrel{+}= v$\\\scriptsize (overflow guarded)};
    \node[fcdec, below=of add] (th) {$b \geq 50$?};
    \node[fcproc, below left=0.9cm and -0.6cm of th] (col) {enter COLLECTING\\\scriptsize RLED ON, BLED OFF};
    \node[fcproc, below right=0.9cm and -0.6cm of th] (ready) {enter READY\\\scriptsize RLED OFF, BLED ON};
    \node[fcterm, below=3.0cm of th] (e) {return};

    \draw[fa] (s) -- (idle);
    \draw[fa] (idle) -- node[right, font=\scriptsize]{yes} (add);
    \draw[fa] (idle) -- node[above, font=\scriptsize]{no} (rdy);
    \draw[fa] (rdy) -- node[above, font=\scriptsize]{no} (rej);
    \draw[fa] (rdy) -- node[right, font=\scriptsize]{yes} (add2);
    \draw[fa] (add) -- (th);
    \draw[fa] (th) -| node[above, font=\scriptsize, pos=0.2]{no} (col);
    \draw[fa] (th) -| node[above, font=\scriptsize, pos=0.2]{yes} (ready);
    \draw[fa] (col) |- (e);
    \draw[fa] (ready) |- (e);
    \draw[fa] (add2) |- (e);
  \end{tikzpicture}}
  \caption{Coin processing in the FSM}
  \label{fig:flow-coin}
\end{figure}

\noindent On the board, coins come from K1 gestures and are first \emph{staged} by the front panel
(Figure~\ref{fig:flow-staging}); a real coin acceptor would instead feed \code{hal_coin_pulse}, which
emits COIN($v$) events directly.
